Production System Architecture Guide

Includes PostgreSQL relational structure and Node.js M-Pesa Daraja Callback integration snippets.

1. PostgreSQL Database Schema (`schema.sql`)

-- PostgreSQL Schema for Ticketsasa Replica Engine

CREATE TABLE events (
    id SERIAL PRIMARY KEY,
    title VARCHAR(255) NOT NULL,
    category VARCHAR(50) NOT NULL,
    venue VARCHAR(255) NOT NULL,
    event_date TIMESTAMP NOT NULL,
    image_url TEXT NOT NULL,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

CREATE TABLE ticket_tiers (
    id SERIAL PRIMARY KEY,
    event_id INT REFERENCES events(id) ON DELETE CASCADE,
    tier_name VARCHAR(50) NOT NULL, -- 'Standard', 'VIP', 'VVIP'
    price_kes DECIMAL(10, 2) NOT NULL,
    available_qty INT NOT NULL
);

CREATE TABLE tickets (
    ticket_uuid VARCHAR(64) PRIMARY KEY,
    event_id INT REFERENCES events(id),
    tier_name VARCHAR(50) NOT NULL,
    customer_name VARCHAR(100) NOT NULL,
    customer_phone VARCHAR(20) NOT NULL,
    customer_email VARCHAR(100) NOT NULL,
    mpesa_receipt VARCHAR(50) UNIQUE,
    is_checked_in BOOLEAN DEFAULT FALSE,
    checked_in_at TIMESTAMP NULL,
    purchase_time TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

2. M-Pesa Daraja STK Push Callback Express Handler (`mpesaController.js`)

// Express JS Handler for M-Pesa STK Push Webhook
const express = require('express');
const router = express.Router();

router.post('/api/v1/mpesa/callback', async (req, res) => {
    try {
        const callbackData = req.body.Body.stkCallback;
        
        if (callbackData.ResultCode === 0) {
            const meta = callbackData.CallbackMetadata.Item;
            const mpesaReceipt = meta.find(i => i.Name === 'MpesaReceiptNumber').Value;
            const phoneNumber = meta.find(i => i.Name === 'PhoneNumber').Value;
            const amount = meta.find(i => i.Name === 'Amount').Value;

            // 1. Generate unique ticket token
            const ticketUuid = 'TICKET-' + Math.floor(1000 + Math.random() * 9000) + '-' + Date.now().toString().slice(-4);

            // 2. Insert into Postgres DB
            await db.query(
                `INSERT INTO tickets(ticket_uuid, event_id, tier_name, customer_name, customer_phone, customer_email, mpesa_receipt)
                 VALUES ($1, $2, $3, $4, $5, $6, $7)`,
                [ticketUuid, req.query.eventId, req.query.tier, req.query.name, phoneNumber, req.query.email, mpesaReceipt]
            );

            // 3. Trigger SMS / Email QR Code Dispatch (Twilio / Sendgrid / Africa's Talking)
            console.log(`Ticket ${ticketUuid} issued successfully for M-Pesa Ref: ${mpesaReceipt}`);
        }
        res.status(200).json({ ResultCode: 0, ResultDesc: "Accepted" });
    } catch (err) {
        console.error('Error processing M-Pesa callback:', err);
        res.status(500).send("Internal Server Error");
    }
});
